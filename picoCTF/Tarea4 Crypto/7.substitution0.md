key = "OHNFUMWSVZLXEGCPTAJDYIRKQB"
alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
cipher_text = "pvncNDM{5YH5717Y710G_3I0XY710G_03055505}"

def decrypt(text, key, alphabet):
    decoded = ""
    for char in text:
        if char.upper() in key:
            # Encuentra la posición en la clave y busca la letra original
            idx = key.index(char.upper())
            original_char = alphabet[idx]
            # Mantiene mayúsculas/minúsculas originales
            decoded += original_char if char.isupper() else original_char.lower()
        else:
            decoded += char
    return decoded

print(decrypt(cipher_text, key, alphabet))
