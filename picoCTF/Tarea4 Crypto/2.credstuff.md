from string import ascii_lowercase, ascii_uppercase

# La contraseña que acabas de encontrar con 'sed'
enc_flag = 'cvpbPGS{P7e1S_54I35_71Z3}'

for shift in range(26):
    flag = ''
    for char in enc_flag:
        if char in ascii_lowercase:
            index = ascii_lowercase.index(char)
            flag += ascii_lowercase[(index + shift) % 26]
        elif char in ascii_uppercase:
            index = ascii_uppercase.index(char)
            flag += ascii_uppercase[(index + shift) % 26]
        else:
            flag += char

    # El desplazamiento 13 es el estándar "ROT13"
    if shift == 13:
        print(f"¡ENCONTRADA! Shift {shift}: {flag}")
